### 3. MinimapManager.cs

```csharp
using System.Collections.Generic;
using UnityEngine;
using UnityGameFramework.Runtime;
using GameFramework;
using GameFramework.Event;

namespace AAAGame.MiniMap
{
    public class MinimapManager : GameFrameworkComponent
    {
        [SerializeField] private MinimapConfig config = new MinimapConfig();

        private Dictionary<int, MinimapUnitData> units = new Dictionary<int, MinimapUnitData>();
        private List<MinimapUnitData> unitsList = new List<MinimapUnitData>();
        private bool[,] fogVisibility;
        private int nextUnitId = 1;

        public MinimapConfig Config => config;

        protected override void Awake()
        {
            base.Awake();
            
            if (config.EnableFogOfWar)
            {
                fogVisibility = new bool[config.FogGridSize, config.FogGridSize];
            }
            
            Log.Info("[Minimap] MinimapManager initialized");
        }

        private void LateUpdate()
        {
            if (config.EnableFogOfWar)
            {
                ProcessFogOfWar();
            }

            BroadcastMinimapUpdate();
        }

        public int RegisterUnit(Vector3 worldPosition, SideType side)
        {
            int unitId = nextUnitId++;
            var unitData = new MinimapUnitData(unitId, worldPosition, side, true);
            units[unitId] = unitData;
            return unitId;
        }

        public void UpdateUnit(int unitId, Vector3 worldPosition, SideType side)
        {
            if (units.ContainsKey(unitId))
            {
                var unitData = units[unitId];
                unitData.WorldPosition = worldPosition;
                unitData.Side = side;
                units[unitId] = unitData;
            }
        }

        public void UnregisterUnit(int unitId)
        {
            units.Remove(unitId);
        }

        private void ProcessFogOfWar()
        {
            if (fogVisibility == null) return;

            for (int x = 0; x < config.FogGridSize; x++)
            {
                for (int z = 0; z < config.FogGridSize; z++)
                {
                    fogVisibility[x, z] = false;
                }
            }

            foreach (var unit in units.Values)
            {
                if (unit.Side == SideType.PlayerSide)
                {
                    Vector2Int gridPos = WorldPosToGrid(unit.WorldPosition);
                    IlluminateGridArea(gridPos, config.VisionRadius);
                }
            }

            var updatedUnits = new Dictionary<int, MinimapUnitData>();
            foreach (var kvp in units)
            {
                var unit = kvp.Value;
                Vector2Int gridPos = WorldPosToGrid(unit.WorldPosition);
                
                if (unit.Side == SideType.PlayerSide)
                {
                    unit.IsVisible = true;
                }
                else
                {
                    unit.IsVisible = IsGridPosVisible(gridPos);
                }
                
                updatedUnits[kvp.Key] = unit;
            }
            
            units = updatedUnits;
        }

        private void IlluminateGridArea(Vector2Int center, float radius)
        {
            int radiusInt = Mathf.CeilToInt(radius);
            
            for (int x = center.x - radiusInt; x <= center.x + radiusInt; x++)
            {
                for (int z = center.y - radiusInt; z <= center.y + radiusInt; z++)
                {
                    if (x >= 0 && x < config.FogGridSize && z >= 0 && z < config.FogGridSize)
                    {
                        float distance = Vector2.Distance(center, new Vector2(x, z));
                        if (distance <= radius)
                        {
                            fogVisibility[x, z] = true;
                        }
                    }
                }
            }
        }

        private Vector2Int WorldPosToGrid(Vector3 worldPos)
        {
            float normalizedX = Mathf.InverseLerp(config.WorldMinX, config.WorldMaxX, worldPos.x);
            float normalizedZ = Mathf.InverseLerp(config.WorldMinZ, config.WorldMaxZ, worldPos.z);
            
            int gridX = Mathf.Clamp(Mathf.FloorToInt(normalizedX * config.FogGridSize), 0, config.FogGridSize - 1);
            int gridZ = Mathf.Clamp(Mathf.FloorToInt(normalizedZ * config.FogGridSize), 0, config.FogGridSize - 1);
            
            return new Vector2Int(gridX, gridZ);
        }

        private bool IsGridPosVisible(Vector2Int gridPos)
        {
            if (gridPos.x < 0 || gridPos.x >= config.FogGridSize || gridPos.y < 0 || gridPos.y >= config.FogGridSize)
                return false;
            
            return fogVisibility[gridPos.x, gridPos.y];
        }

        private void BroadcastMinimapUpdate()
        {
            unitsList.Clear();
            unitsList.AddRange(units.Values);

            GameEventArgs e = MinimapUpdateEventArgs.Create(unitsList, fogVisibility);
            GF.Event.Fire(this, e);
            ReferencePool.Release(e);
        }

        public int GetUnitCount()
        {
            return units.Count;
        }
    }
}
```

### 4. MinimapReportComponent.cs

```csharp
using UnityEngine;

namespace AAAGame.MiniMap
{
    /// <summary>
    /// 小地图报告组件
    /// 挂在需要显示在小地图上的单位身上
    /// </summary>
    public class MinimapReportComponent : MonoBehaviour
    {
        private MinimapManager minimapManager;
        private int unitId = -1;
        private SideType side = SideType.NoSide;

        /// <summary>
        /// 初始化
        /// </summary>
        public void Initialize(SideType unitSide)
        {
            side = unitSide;
            minimapManager = GameEntry.GetComponent<MinimapManager>();
            
            if (minimapManager != null)
            {
                unitId = minimapManager.RegisterUnit(transform.position, side);
            }
        }

        /// <summary>
        /// Tick 更新（由 Entity 调用）
        /// </summary>
        public void Tick()
        {
            if (minimapManager != null && unitId >= 0)
            {
                minimapManager.UpdateUnit(unitId, transform.position, side);
            }
        }

        private void OnDestroy()
        {
            if (minimapManager != null && unitId >= 0)
            {
                minimapManager.UnregisterUnit(unitId);
            }
        }
    }
}
```

### 5. MinimapUI.cs

```csharp
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.UI;
using GameFramework;
using GameFramework.Event;
using UnityGameFramework.Runtime;

namespace AAAGame.MiniMap
{
    /// <summary>
    /// 小地图 UI 显示
    /// 继承 UIFormBase，监听更新事件并显示单位点
    /// </summary>
    public class MinimapUI : UIFormBase
    {
        [Header("UI 组件")]
        [SerializeField] private RectTransform minimapContainer;
        [SerializeField] private GameObject unitDotPrefab;
        [SerializeField] private RawImage fogMaskImage;

        [Header("小地图尺寸")]
        [SerializeField] private float minimapSize = 200f;

        private MinimapManager minimapManager;
        private Dictionary<int, Image> unitDots = new Dictionary<int, Image>();
        private List<int> unitsToRemove = new List<int>();
        private Texture2D fogTexture;

        protected override void OnInit(object userData)
        {
            base.OnInit(userData);
            
            minimapManager = GameEntry.GetComponent<MinimapManager>();
            
            if (minimapManager == null)
            {
                Log.Error("[MinimapUI] MinimapManager not found!");
                return;
            }

            // 订阅更新事件
            GFBuiltin.Event.Subscribe(MinimapUpdateEventArgs.EventId, OnMinimapUpdate);
            
            // 初始化战争迷雾纹理
            if (minimapManager.Config.EnableFogOfWar && fogMaskImage != null)
            {
                int gridSize = minimapManager.Config.FogGridSize;
                fogTexture = new Texture2D(gridSize, gridSize, TextureFormat.RGBA32, false);
                fogTexture.filterMode = FilterMode.Point;
                fogMaskImage.texture = fogTexture;
            }
            
            Log.Info("[MinimapUI] MinimapUI initialized");
        }

        protected override void OnOpen(object userData)
        {
            base.OnOpen(userData);
        }

        protected override void OnClose(bool isShutdown, object userData)
        {
            base.OnClose(isShutdown, userData);
            
            // 清理所有点
            foreach (var dot in unitDots.Values)
            {
                if (dot != null)
                {
                    Destroy(dot.gameObject);
                }
            }
            unitDots.Clear();
        }

        protected override void OnRecycle()
        {
            base.OnRecycle();
            
            // 取消订阅事件
            if (GFBuiltin.Event != null)
            {
                GFBuiltin.Event.Unsubscribe(MinimapUpdateEventArgs.EventId, OnMinimapUpdate);
            }
            
            // 清理纹理
            if (fogTexture != null)
            {
                Destroy(fogTexture);
                fogTexture = null;
            }
        }

        /// <summary>
        /// 小地图更新事件处理
        /// </summary>
        private void OnMinimapUpdate(object sender, GameEventArgs e)
        {
            MinimapUpdateEventArgs ne = (MinimapUpdateEventArgs)e;
            
            if (ne.Units == null) return;

            // 标记所有现有单位为待移除
            unitsToRemove.Clear();
            unitsToRemove.AddRange(unitDots.Keys);

            // 更新或创建单位点
            foreach (var unit in ne.Units)
            {
                // 如果不可见（战争迷雾），跳过
                if (!unit.IsVisible) continue;

                // 如果已存在，更新位置和颜色
                if (unitDots.ContainsKey(unit.UnitId))
                {
                    UpdateUnitDot(unit);
                    unitsToRemove.Remove(unit.UnitId);
                }
                // 否则创建新点
                else
                {
                    CreateUnitDot(unit);
                }
            }

            // 移除不再存在的单位点
            foreach (int unitId in unitsToRemove)
            {
                RemoveUnitDot(unitId);
            }

            // 更新战争迷雾
            if (ne.FogVisibility != null && fogTexture != null)
            {
                UpdateFogMask(ne.FogVisibility);
            }
        }

        /// <summary>
        /// 创建单位点
        /// </summary>
        private void CreateUnitDot(MinimapUnitData unit)
        {
            if (unitDotPrefab == null || minimapContainer == null) return;

            GameObject dotObj = Instantiate(unitDotPrefab, minimapContainer);
            Image dotImage = dotObj.GetComponent<Image>();
            
            if (dotImage == null)
            {
                dotImage = dotObj.AddComponent<Image>();
            }

            // 设置颜色
            dotImage.color = minimapManager.Config.GetSideColor(unit.Side);

            // 设置位置
            RectTransform dotRect = dotObj.GetComponent<RectTransform>();
            Vector2 minimapPos = WorldToMinimapPosition(unit.WorldPosition);
            dotRect.anchoredPosition = minimapPos;

            unitDots[unit.UnitId] = dotImage;
        }

        /// <summary>
        /// 更新单位点
        /// </summary>
        private void UpdateUnitDot(MinimapUnitData unit)
        {
            if (!unitDots.ContainsKey(unit.UnitId)) return;

            Image dotImage = unitDots[unit.UnitId];
            if (dotImage == null) return;

            // 更新颜色
            dotImage.color = minimapManager.Config.GetSideColor(unit.Side);

            // 更新位置
            RectTransform dotRect = dotImage.GetComponent<RectTransform>();
            Vector2 minimapPos = WorldToMinimapPosition(unit.WorldPosition);
            dotRect.anchoredPosition = minimapPos;
        }

        /// <summary>
        /// 移除单位点
        /// </summary>
        private void RemoveUnitDot(int unitId)
        {
            if (!unitDots.ContainsKey(unitId)) return;

            Image dotImage = unitDots[unitId];
            if (dotImage != null)
            {
                Destroy(dotImage.gameObject);
            }

            unitDots.Remove(unitId);
        }

        /// <summary>
        /// 世界坐标转小地图坐标
        /// </summary>
        private Vector2 WorldToMinimapPosition(Vector3 worldPos)
        {
            if (minimapManager == null) return Vector2.zero;

            MinimapConfig config = minimapManager.Config;

            // 归一化世界坐标 (0-1)
            float normalizedX = Mathf.InverseLerp(config.WorldMinX, config.WorldMaxX, worldPos.x);
            float normalizedZ = Mathf.InverseLerp(config.WorldMinZ, config.WorldMaxZ, worldPos.z);

            // 转换为小地图坐标（中心为原点）
            float minimapX = (normalizedX - 0.5f) * minimapSize;
            float minimapY = (normalizedZ - 0.5f) * minimapSize;

            return new Vector2(minimapX, minimapY);
        }

        /// <summary>
        /// 更新战争迷雾遮罩
        /// </summary>
        private void UpdateFogMask(bool[,] fogVisibility)
        {
            if (fogTexture == null) return;

            int gridSize = minimapManager.Config.FogGridSize;

            for (int x = 0; x < gridSize; x++)
            {
                for (int z = 0; z < gridSize; z++)
                {
                    // 可见区域：透明，不可见区域：黑色半透明
                    Color color = fogVisibility[x, z] ? Color.clear : new Color(0, 0, 0, 0.7f);
                    fogTexture.SetPixel(x, z, color);
                }
            }

            fogTexture.Apply();
        }
    }
}
```

## 使用说明

### 1. 场景设置

1. 在场景中创建一个空 GameObject，命名为 `MinimapManager`
2. 添加 `MinimapManager` 组件
3. 配置参数：
   - 世界坐标范围（WorldMinX, WorldMaxX, WorldMinZ, WorldMaxZ）
   - 势力颜色（PlayerColor, EnemyColor）
   - 战争迷雾设置（可选）

### 2. 单位设置

在需要显示在小地图上的单位 Prefab 上：

```csharp
// 在单位的初始化代码中
MinimapReportComponent reportComp = gameObject.AddComponent<MinimapReportComponent>();
reportComp.Initialize(SideType.PlayerSide); // 或 EnemySide

// 在单位的 Update 中
reportComp.Tick();
```

### 3. UI 设置

1. 创建 UI Canvas
2. 添加小地图 UI 面板（正方形）
3. 添加 `MinimapUI` 组件
4. 配置：
   - MinimapContainer: 小地图容器 RectTransform
   - UnitDotPrefab: 单位点预制体（小圆点 Image）
   - FogMaskImage: 战争迷雾遮罩（可选）
   - MinimapSize: 小地图尺寸（像素）

## 数据流

```
单位 Update
    ↓
MinimapReportComponent.Tick()
    ↓
MinimapManager.UpdateUnit()
    ↓
MinimapManager.LateUpdate()
    ↓
计算战争迷雾（可选）
    ↓
广播 MinimapUpdateEventArgs
    ↓
MinimapUI.OnMinimapUpdate()
    ↓
更新点位置和颜色
```

## 特性

✅ 基础功能
- 区分势力（玩家/敌方）
- 实时位置更新
- 可配置世界坐标范围
- 可配置势力颜色

⏳ 进阶功能（优先级低）
- 格子战争迷雾
- 视野范围计算
- 迷雾遮罩显示

## 注意事项

1. MinimapManager 必须作为 GameFrameworkComponent 挂在场景中
2. MinimapReportComponent 需要手动调用 Initialize() 和 Tick()
3. 单位点预制体建议使用小圆点 Image（5x5 像素）
4. 战争迷雾功能会增加性能开销，建议在需要时才启用
